function processDecodedText(text) { const now = Date.now(); /* 같은 QR 반복 인식 방지 */ if (text === lastText && now - lastScanTime < 3000) { return; } lastText = text; lastScanTime = now; successSignal(); /* 51~63번째 문자 추출 */ const extractedText = text.length >= 63 ? text.substring(50, 63) : "추출 불가 (문자열 길이 부족)"; /* 스캔 시간 */ const scanTime = new Date().toLocaleString("ko-KR", { year:"numeric", month:"2-digit", day:"2-digit", hour:"2-digit", minute:"2-digit", second:"2-digit", hour12:false }); /* 최근 처리 결과 표시 */ document.getElementById("recentResult").textContent = `최근 처리: ${scanTime} | 추출값: ${extractedText}`; setMainStatus("QR 코드 인식 완료"); showResultStatus("전송 중"); sendToSheet(text); }
